

import Link from "next/link";
import { majorArcanaMeanings } from "@/data/meanings/major-arcana";

export default function MajorMeaningCheckPage() {
  const expectedIds = Array.from(
    { length: 22 },
    (_, index) => `major-${String(index).padStart(2, "0")}`,
  );

  const actualIds = majorArcanaMeanings.map((meaning) => meaning.cardId);

  const missingIds = expectedIds.filter((id) => !actualIds.includes(id));

  const duplicateIds = actualIds.filter(
    (id, index) => actualIds.indexOf(id) !== index,
  );

  const isComplete =
    majorArcanaMeanings.length === 22 &&
    missingIds.length === 0 &&
    duplicateIds.length === 0;

  return (
    <main className="min-h-screen bg-[#f8f5ef] px-6 py-12 text-[#29251f]">
      <div className="mx-auto max-w-3xl">
        <Link
          href="/"
          className="text-sm text-[#76634c] underline"
        >
          ← กลับหน้าหลัก
        </Link>

        <h1 className="mt-6 text-3xl font-bold">
          ตรวจสอบความหมายไพ่ Major Arcana
        </h1>

        <p className="mt-3 text-gray-600">
          ตรวจสอบจำนวนไพ่ รหัสที่ขาดหาย และรหัสที่ซ้ำกัน
        </p>

        <section className="mt-8 rounded-xl border border-gray-200 bg-white p-6">
          <p className="text-lg">
            จำนวนรายการความหมาย:{" "}
            <strong>{majorArcanaMeanings.length} / 22</strong>
          </p>

          <p className="mt-3">
            รหัสไพ่ไม่ซ้ำกัน:{" "}
            <strong>{duplicateIds.length === 0 ? "ใช่" : "ไม่ใช่"}</strong>
          </p>

          <p className="mt-3">
            รหัสไพ่ครบถ้วน:{" "}
            <strong>{missingIds.length === 0 ? "ใช่" : "ไม่ใช่"}</strong>
          </p>

          <div
            className={`mt-5 rounded-lg p-4 ${
              isComplete
                ? "bg-green-100 text-green-800"
                : "bg-red-100 text-red-800"
            }`}
          >
            <strong>
              {isComplete
                ? "ผ่าน: ความหมายไพ่ครบ 22 ใบ และไม่มีรหัสซ้ำ"
                : "ยังไม่ผ่าน: กรุณาตรวจสอบรายการด้านล่าง"}
            </strong>
          </div>

          {missingIds.length > 0 && (
            <div className="mt-5">
              <h2 className="font-bold">รหัสที่ขาดหาย</h2>
              <p className="mt-2">{missingIds.join(", ")}</p>
            </div>
          )}

          {duplicateIds.length > 0 && (
            <div className="mt-5">
              <h2 className="font-bold">รหัสที่ซ้ำกัน</h2>
              <p className="mt-2">
                {[...new Set(duplicateIds)].join(", ")}
              </p>
            </div>
          )}
        </section>

        <section className="mt-8 rounded-xl border border-gray-200 bg-white p-6">
          <h2 className="text-xl font-bold">รายการรหัสไพ่</h2>

          <div className="mt-4 grid grid-cols-2 gap-3 sm:grid-cols-3">
            {majorArcanaMeanings.map((meaning) => (
              <div
                key={meaning.cardId}
                className="rounded-lg border border-gray-200 p-3"
              >
                <p className="font-semibold">{meaning.cardId}</p>
                <p className="mt-1 text-sm text-gray-600">
                  {meaning.keywords.join(" · ")}
                </p>
              </div>
            ))}
          </div>
        </section>
      </div>
    </main>
  );
}
